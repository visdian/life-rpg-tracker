<script>
    // 页面跳转函数
    function navigate(pageId) {
        document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
        document.getElementById(pageId).classList.add('active');
        window.scrollTo(0,0);
    }

    // 数据操作逻辑
    let habits = JSON.parse(localStorage.getItem('lifeRpgData')) || [];

    function renderHabits() {
        const list = document.getElementById('habitList');
        list.innerHTML = '';
        
        habits.forEach(h => {
            const div = document.createElement('div');
            div.className = 'quest-item';
            div.innerHTML = `
                <div>
                    <strong style="display:block; font-size:1.1rem;">${h.name}</strong>
                    <span style="color:#888; font-size:0.85rem;">目标：${h.target}</span>
                </div>
                <div style="color:var(--primary); font-weight:bold;">¥ ${h.reward}</div>
            `;
            list.appendChild(div);
        });

        // 刷新顶部的数值
        const total = habits.reduce((sum, h) => sum + parseInt(h.reward), 0);
        document.getElementById('balanceDisplay').innerText = `¥ ${total}`;
        document.getElementById('lvDisplay').innerText = Math.floor(Math.sqrt(total/100)) || 1;
    }

    function saveAndReturn() {
        const name = document.getElementById('habitName').value;
        const target = document.getElementById('habitTarget').value;
        const reward = document.getElementById('habitReward').value;

        if(name && target && reward) {
            // 存入数组
            habits.push({ name, target, reward });
            // 永久存入浏览器
            localStorage.setItem('lifeRpgData', JSON.stringify(habits));
            
            // 重新渲染主页
            renderHabits();
            
            // 清空表单
            document.getElementById('habitName').value = '';
            document.getElementById('habitTarget').value = '';
            document.getElementById('habitReward').value = '';

            // 关键：跳转回主页
            navigate('homePage');
        } else {
            alert("请完整填写任务信息！");
        }
    }

    // 初始加载
    renderHabits();
</script>
